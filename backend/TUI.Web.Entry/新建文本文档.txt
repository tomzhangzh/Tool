    const table = $(".vp-table").get(0);
    const rows = table.querySelectorAll('tr');

    const attributes = [];

    for (let i = 1; i < rows.length; i++) {
        const cells = rows[i].querySelectorAll('td');
        const attribute = {
            name: cells[0].textContent.trim(),
            label: cells[1].textContent.trim(),
            type: cells[2].textContent.trim(),
            dvalue: cells[3].textContent.trim(),
        };
        attributes.push(attribute);
    }

    var result = Enumerable.from(attributes)
.where(x => x.label.indexOf('原生')<0)
.select(x => {
        var item =
        {
            "component": "t-input", "modelname": "options.comoptions." + x.name,
            "options": {
                "labeloptions": {
                    "label": x.label
                }
            }
        };
if (x.name.indexOf('value')>=0){
item.modelname="modelname";
}
        if (x.type.indexOf('boolean')>=0){
            item.component="t-switch";
        } else if (x.type.indexOf('enum') >= 0) {
            item.component = "t-select";
            item.options.optionValues="";
        }

        return item;
    }).toArray();
    JSON.stringify(result, null, "\t")
