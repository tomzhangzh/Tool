            // 表单校验：扫描容器内动态组件节点的 __dynValidate 钩子
            validateForm: function () {
                var errs = [];
                var nodes = element.querySelectorAll('*');
                for (var i = 0; i < nodes.length; i++) {
                    var n = nodes[i];
                    if (typeof n.__dynValidate === 'function') {
                        try {
                            var ok = n.__dynValidate();
                            if (ok !== true) {
                                var msg = null;
                                if (n.__dynErrors && n.__dynErrors.length) {
                                    msg = n.__dynErrors.map(function (e) { return (e && e.message) || e; }).join('；');
                                }
                                if (!msg) {
                                    try {
                                        var inst = n.__vueParentComponent;
                                        var comp = inst && inst.proxy;
                                        if (comp && comp.comInnerInfo && comp.comInnerInfo.validation && comp.comInnerInfo.validation.errors) {
                                            comp.comInnerInfo.validation.errors.forEach(function (e) { if (e && e.message) msg = e.message; });
                                        }
                                    } catch (e) { }
                                }
                                if (!msg) msg = '存在未通过校验的字段';
                                if (errs.indexOf(msg) < 0) errs.push(msg);
                            }
                        } catch (e) { console.warn('[detail validate]', e); }
                    }
                }
                return errs;
            },
            // 命令入口：newForm（新增，静态 default + preFill 浅合并，不改 PageSetting）/ editForm（回填）
            handleCommand: function (cmd, payload) {
                if (cmd === 'newForm') {
                    var preFill = (payload && payload.preFill) || {};
                    this.form = Object.assign({}, JSON.parse(JSON.stringify(this.__initForm)), preFill);
                    this.formError = '';
                    return Promise.resolve(null);
                }
                if (cmd === 'editForm') {
                    var row = (payload && payload.row) || {};
                    this.form = Object.assign({}, JSON.parse(JSON.stringify(this.__initForm)), row);
                    this.formError = '';
                    return Promise.resolve(null);
                }
                return Promise.resolve(null);
            },
            // 保存：校验 → 业务库 insert/update（body {table,data}，统一走 DynCall）→ 发 detail.saved 事件
            onSave: function () {
                var self = this;
                var errs = this.validateForm();
                if (errs.length) {
                    this.formError = errs.join('；');
                    dyn.showMessage(errs.join('；'), 'warning');
                    return;
                }
                this.loading = true;
                var isEdit = this.form && this.form.Id !== null && this.form.Id !== undefined && this.form.Id !== '';
                // 接口风格适配：/api/platform/* 为平台直连（save?table= 直接发行数据）；否则业务 dyndata（insert/update，body {table,data}）
                var isPlat = (this.addUrl || '').indexOf('/api/platform/') === 0;
                var body = isPlat ? this.form : { table: this.table, data: this.form, project: this.project };
                DynCall.post(isEdit ? this.editUrl : this.addUrl, body, { scopeEl: element })
                    .then(function () {
                        self.loading = false;
                        self.formError = '';
                        dyn.showMessage('保存成功', 'success');
                        var msg = __ctx.msgCenter;
                        if (msg) msg.emitEvent('detail.saved', { action: isEdit ? 'edit' : 'add', row: self.form });
                        // 弹窗模式：通知宿主页面刷新 + 关闭弹窗（宿主据此重新加载列表）
                        if (self.modal) {
                            try {
                                window.dispatchEvent(new CustomEvent('dyndata.saved', { detail: { table: self.table, project: self.project } }));
                            } catch (e) { }
                            var layero = element.closest ? element.closest('.layui-layer') : null;
                            if (window.layui && window.layui.layer) {
                                var times = layero ? layero.getAttribute('times') : null;
                                if (times && times !== '') window.layui.layer.close(times);
                                else window.layui.layer.closeAll();
                            }
                        }
                    })
                    .catch(function (e) { self.loading = false; /* DynCall 已统一 toast */ });
            },
            // 取消：清空表单（新增态骨架）
            onCancel: function () {
                this.form = JSON.parse(JSON.stringify(this.__initForm));
                this.formError = '';
            },
            // 删除：仅编辑态（有主键）可用。确认后 POST deleteUrl {table, keys:{Id}, project}，成功后发 detail.saved{action:'delete'}。
            // 统一走 DynCall（信封解包 + 统一 toast）；blockId 判断用注册时的 'detail' 别名，宿主据此刷新（树/列表）。
            onDelete: function () {
                var self = this;
                var id = this.form[this.keyField];
                if (id === null || id === undefined || id === '' || id === 0) { dyn.showMessage('请先选择要删除的记录', 'warning'); return; }
                if (!this.deleteUrl) { dyn.showMessage('未配置删除接口 deleteUrl', 'warning'); return; }
                var doDelete = function () {
                    self.loading = true;
                    // 接口风格适配：平台直连 delete 用 query 传 table/id；业务 dyndata delete 用 body {table, keys}
                    var isPlat = (self.deleteUrl || '').indexOf('/api/platform/') === 0;
                    var dUrl = isPlat ? (self.deleteUrl + '?table=' + encodeURIComponent(self.table) + '&id=' + encodeURIComponent(id)) : self.deleteUrl;
                    var dBody = isPlat ? {} : { table: self.table, keys: { Id: id }, project: self.project };
                    DynCall.post(dUrl, dBody, { scopeEl: element })
                        .then(function () {
                            self.loading = false;
                            dyn.showMessage('删除成功', 'success');
                            self.form = JSON.parse(JSON.stringify(self.__initForm));
                            self.formError = '';
                            var msg = __ctx.msgCenter;
                            if (msg) msg.emitEvent('detail.saved', { action: 'delete', row: { Id: id } });
                            if (self.modal) {
                                try {
                                    window.dispatchEvent(new CustomEvent('dyndata.saved', { detail: { table: self.table, project: self.project } }));
                                } catch (e) { }
                                var layero = element.closest ? element.closest('.layui-layer') : null;
                                if (window.layui && window.layui.layer) {
                                    var times = layero ? layero.getAttribute('times') : null;
                                    if (times && times !== '') window.layui.layer.close(times);
                                    else window.layui.layer.closeAll();
                                }
                            }
                        })
                        .catch(function (e) { self.loading = false; /* DynCall 已统一 toast */ });
                };
                if (window.dyn && typeof window.dyn.confirmAsync === 'function') {
                    window.dyn.confirmAsync('确定删除该记录吗？').then(function (ok) { if (ok) doDelete(); });
                } else { doDelete(); }
            },
            // 注册本 Block 的保存/取消/删除动作（dyn-click 委托在 document 捕获层触发，ctx.vm 即本 Block proxy）
            registerActions: function () {
                if (!dyn || typeof dyn.defineAction !== 'function') return;
                dyn.defineAction('DetailSave', function (ctx) {
                    var vm = ctx && ctx.vm;
                    if (vm && typeof vm.onSave === 'function') return vm.onSave();
                });
                dyn.defineAction('DetailCancel', function (ctx) {
                    var vm = ctx && ctx.vm;
                    if (vm && typeof vm.onCancel === 'function') return vm.onCancel();
                });
                dyn.defineAction('DetailDelete', function (ctx) {
                    var vm = ctx && ctx.vm;
                    if (vm && typeof vm.onDelete === 'function') return vm.onDelete();
                });
                // 注意：无需 dyn.rebindActions —— generateSelector 每次点击动态生成，动作注册即生效；
                // 多次 rebind 会导致 document 监听叠加（一次点击触发多次动作）。
            }
        },
        mounted: function () {
            this.registerActions();
            // 向根 app 消息中心注册本积木（接收 newForm/editForm 命令）
            var self = this, msg = __ctx.msgCenter;
            if (msg && typeof msg.registerBlock === 'function') {
                msg.registerBlock('detail', function (cmd, payload) { return self.handleCommand(cmd, payload); });
            }
        }
    };
    </script>
